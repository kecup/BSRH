import torch
import numpy as np

def evaluate_step_model(model, test_data, input_cols, simulate_motor_move):
    """
    Evaluates a step-predicting model using average beam-target error.

    Args:
        model: trained PyTorch model (predicting delta steps)
        test_data: list of dicts with fields like CamX, CamY, MotX, MotY, TgtX, TgtY
        input_cols: list of input feature column names for the model
        simulate_motor_move: function(cam_pos, motor_pos, delta_steps) -> new_cam_pos

    Returns:
        avg_error: average distance between final beam position and target
    """
    model.eval()
    total_error = 0.0
    count = 0

    for sample in test_data:
        # Get current state
        cam_pos = np.array([sample["CamX"], sample["CamY"]])
        motor_pos = np.array([sample["MotX"], sample["MotY"]])
        target_pos = np.array([sample["TgtX"], sample["TgtY"]])

        # Prepare model input
        input_vals = np.array([sample[col] for col in input_cols], dtype=np.float32)
        input_tensor = torch.tensor(input_vals).unsqueeze(0)  # shape: [1, input_dim]

        # Predict delta motor steps
        with torch.no_grad():
            delta_steps = model(input_tensor).squeeze(0).numpy()

        # Simulate the movement and get new camera position
        new_cam_pos = simulate_motor_move(cam_pos, motor_pos, delta_steps)

        # Compute beam-target error
        error = np.linalg.norm(new_cam_pos - target_pos)
        total_error += error
        count += 1

    avg_error = total_error / count
    return avg_error
