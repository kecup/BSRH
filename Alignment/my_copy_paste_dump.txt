for idx, (run_num, run_data) in enumerate(data.items()):
    ax = axes[idx]
    events = run_data['events']
    cpc = run_data['cpc']
    
    for group_idx, mask in enumerate(masks[run_num]):
        ax.scatter(
            [events[i] for i in mask],  # x-values
            [cpc[i] for i in mask],     # y-values
            color=colors[group_idx],
            label=f'Group {group_idx + 1}'
        )
    
    ax.set_title(f'Run {run_num}')
    ax.set_xlabel('Events')
    ax.set_ylabel('CPC')
    ax.legend()
