from datetime import datetime, timedelta
import sys

# Current time
now = datetime.now()

# Target time = tomorrow 10:00 AM
tomorrow = now.date() + timedelta(days=1)
target_time = datetime.combine(tomorrow, datetime.strptime("10:00", "%H:%M").time())

# Compare
if now > target_time:
    print("It's later than tomorrow 10:00 AM. Stopping script.")
    sys.exit()

print("Still before tomorrow 10:00 AM. Continue running...")
