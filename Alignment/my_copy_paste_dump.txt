import pandas as pd

# Example DataFrame
df = pd.DataFrame({
    "t2X": [10, 15, 30, 45],
    "t2Y": [5, 25, 35, 50]
})

# Define ranges
x_min, x_max = 10, 40
y_min, y_max = 10, 40

# Filter rows
filtered_df = df[
    (df["t2X"].between(x_min, x_max)) &
    (df["t2Y"].between(y_min, y_max))
]

print(filtered_df)
