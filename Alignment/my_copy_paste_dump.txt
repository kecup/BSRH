import pandas as pd
import matplotlib.pyplot as plt

df = pd.read_csv("your_file.csv")

plt.figure(figsize=(6,6))
plt.scatter(df["xA"], df["xB"], alpha=0.6, label="X comparison")
plt.scatter(df["yA"], df["yB"], alpha=0.6, label="Y comparison")
plt.plot([df[["xA","xB","yA","yB"]].min().min(), 
          df[["xA","xB","yA","yB"]].max().max()],
         [df[["xA","xB","yA","yB"]].min().min(), 
          df[["xA","xB","yA","yB"]].max().max()],
         'r--', label="Perfect match")
plt.xlabel("Method A")
plt.ylabel("Method B")
plt.legend()
plt.title("Coordinate Comparison (A vs B)")
plt.show()


df["dx"] = df["xA"] - df["xB"]
df["dy"] = df["yA"] - df["yB"]

df[["dx","dy"]].hist(bins=30)
plt.suptitle("Error Distribution (A - B)")
plt.show()
