import pandas as pd

# Load your data
df = pd.read_csv("your_file.csv")  # adjust filename

# Group by motor positions to see how much spread there is in camera readings
spread = df.groupby(["mX", "mY", "mPhi", "mTheta"])[["cam1X", "cam1Y"]].agg(['mean', 'std', 'count'])
print(spread.sort_values(('cam1X', 'std'), ascending=False).head(10))
